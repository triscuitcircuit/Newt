cmake_minimum_required(VERSION 3.20)
project(NewtPackageConsumer LANGUAGES CXX)
find_package(Newt CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE Newt::newt)
if(TARGET Newt::usb)
    target_link_libraries(consumer PRIVATE Newt::usb)
    target_compile_definitions(consumer PRIVATE NEWT_TEST_USB)
endif()
